generate_production_yields.py 
# Run this script to generate Maliban_Production_Yields_1000.csv
import csv, random
from datetime import datetime, timedelta

start_date = datetime(2025, 1, 1)
lines = ['Line_01_Ratmalana', 'Line_02_Ratmalana', 'Line_03_Ratmalana', 'Line_04_Galle']
products = {
    'Line_01_Ratmalana': 'Smart Cream Cracker', 'Line_02_Ratmalana': 'Ginger Biscuit',
    'Line_03_Ratmalana': 'Lemon Puff', 'Line_04_Galle': 'Chocolate Cream'
}
supervisors = ['Perera.D', 'Silva.A', 'Fernando.K', 'Jayasinghe.M']

with open("Maliban_Production_Yields_1000.csv", 'w', newline='') as f:
    w = csv.writer(f)
    w.writerow(['Date', 'Factory_Line', 'Product_Name', 'Target_Batches', 'Actual_Batches', 'Wastage_KG', 'Downtime_Minutes', 'Shift_Supervisor'])
    
    row_count = 0
    day = 0
    while row_count < 1000:
        dt = (start_date + timedelta(days=day)).strftime('%Y-%m-%d')
        for l in lines:
            if row_count >= 1000: break
            p, s = products[l], random.choice(supervisors)
            t = random.randint(140, 210)
            rnd = random.random()
            if rnd < 0.06:
                downtime = random.randint(45, 180)
                act = int(t * random.uniform(0.60, 0.82))
                wastage = round(random.uniform(65.0, 140.0), 1)
            elif rnd > 0.94:
                downtime = 0
                act = int(t * random.uniform(1.01, 1.05))
                wastage = round(random.uniform(5.0, 12.0), 1)
            else:
                downtime = random.choice([0, 0, 5, 10, 15])
                act = int(t * random.uniform(0.95, 1.01))
                wastage = round(random.uniform(12.5, 38.0), 1)
            w.writerow([dt, l, p, t, max(0, act), wastage, downtime, s])
            row_count += 1
        day += 1
print("Done!")

import csv, random

distributors = [
    ('DIST-901', 'Lanka Wholesale Central', 'Western Province (SL)', 'General Trade'),
    ('DIST-902', 'UK EuroFoods Ltd', 'United Kingdom', 'Export'),
    ('DIST-903', 'Aussie Bites & Snacks', 'Australia', 'Export'),
    ('DIST-904', 'Ruhunu Retailers Coop', 'Southern Province (SL)', 'Modern Trade'),
    ('DIST-905', 'Kandy Mega Distributors', 'Central Province (SL)', 'General Trade'),
    ('DIST-906', 'Jaffna Unified Traders', 'Northern Province (SL)', 'General Trade'),
    ('DIST-907', 'Maldives Retail Group', 'Maldives', 'Export'),
    ('DIST-908', 'MiddleEast FMCG LLC', 'UAE', 'Export'),
    ('DIST-909', 'Wayamba Traders', 'North Western Province', 'General Trade'),
    ('DIST-910', 'Sabaragamuwa Distribution', 'Sabaragamuwa', 'General Trade')
]
categories = ['Savoury (Crackers)', 'Sweet (Ginger/Marie)', 'Cream Range']

with open("Maliban_Global_Sales_Analysis.csv", 'w', newline='', encoding='utf-8') as f:
    w = csv.writer(f)
    w.writerow(['Month_Year', 'Distributor_ID', 'Distributor_Name', 'Region_Country', 'Channel', 'Product_Category', 'Units_Shipped', 'Target_Revenue_LKR', 'Actual_Revenue_LKR', 'Revenue_Gap_LKR', 'Outstanding_Balance_LKR', 'Credit_Status'])
    
    for year in range(2023, 2027):
        for month in range(1, 13):
            if year == 2026 and month > 4: break
            dt = f"{year}-{month:02d}-01"
            for d in distributors:
                for c in categories:
                    base = random.uniform(52000000, 115000000) if d[3] == 'Export' else random.uniform(16000000, 44000000)
                    if month in [4, 12]: base *= 1.25 # Seasonal peak
                    perf = random.normalvariate(0.96, 0.08)
                    target = int(round(base, -3))
                    act = int(round(base * perf, -3))
                    gap = target - act
                    units = int(act / random.uniform(220, 370))
                    
                    if perf < 0.85:
                        bal, status = int(act * random.uniform(0.25, 0.45)), 'Hold'
                    elif perf < 0.93:
                        bal, status = int(act * random.uniform(0.10, 0.20)), 'Review'
                    else:
                        bal, status = int(act * random.uniform(0.0, 0.05)), 'Clear'
                    
                    w.writerow([dt, d[0], d[1], d[2], d[3], c, units, target, act, gap, bal, status])

# Run this within Fabric Notebooks
import pandas as pd
import numpy as np
from datetime import datetime
from sklearn.linear_model import LinearRegression
from pyspark.sql import SparkSession

spark = SparkSession.builder.getOrCreate()

# Load delta structures
df_prod = spark.read.table("fact_production_yields").toPandas()
df_sales = spark.read.table("fact_global_sales_analysis").toPandas()

df_prod['Date'] = pd.to_datetime(df_prod['Date'])
df_sales['Month_Year'] = pd.to_datetime(df_sales['Month_Year'])

df_prod['Year_Month'] = df_prod['Date'].dt.to_period('M').astype(str)
df_sales['Year_Month'] = df_sales['Month_Year'].dt.to_period('M').astype(str)

product_mapping = {
    'Smart Cream Cracker': 'Savoury (Crackers)',
    'Ginger Biscuit': 'Sweet (Ginger/Marie)',
    'Lemon Puff': 'Cream Range',
    'Chocolate Cream': 'Cream Range'
}
df_prod['Product_Category'] = df_prod['Product_Name'].map(product_mapping)

monthly_prod = df_prod.groupby(['Year_Month', 'Product_Category']).agg(
    Total_Actual_Batches=('Actual_Batches', 'sum'),
    Total_Wastage_KG=('Wastage_KG', 'sum'),
    Total_Downtime_Minutes=('Downtime_Minutes', 'sum')
).reset_index()

df_merged = pd.merge(df_sales, monthly_prod, on=['Year_Month', 'Product_Category'], how='inner')

# Train Machine Learning Models
df_merged['Forecasted_Revenue_LKR'] = 0.0
for category in df_merged['Product_Category'].unique():
    cat_mask = df_merged['Product_Category'] == category
    df_cat = df_merged[cat_mask].copy()
    if len(df_cat) > 3:
        df_cat['Time_Idx'] = np.arange(len(df_cat))
        X = df_cat[['Time_Idx']]
        y = df_cat['Actual_Revenue_LKR']
        model = LinearRegression().fit(X, y)
        df_merged.loc[cat_mask, 'Forecasted_Revenue_LKR'] = model.predict(X).astype(int)

# Outlier Anomaly calculations
mean_gap = df_merged['Revenue_Gap_LKR'].mean()
std_gap = df_merged['Revenue_Gap_LKR'].std()
df_merged['Operational_Risk_Score'] = ((df_merged['Revenue_Gap_LKR'] - mean_gap) / std_gap).round(2)
df_merged['AI_Gap_Analysis_Flag'] = np.where(
    (df_merged['Operational_Risk_Score'] > 1.0) & (df_merged['Total_Downtime_Minutes'] > df_merged['Total_Downtime_Minutes'].median()), 
    "CRITICAL: Production Bottleneck Linked to Revenue Drop",
    "Normal Operational Variance"
)

# Output Delta
spark_df = spark.createDataFrame(df_merged.astype(str))
spark_df.write.format("delta").mode("overwrite").saveAsTable("ai_maliban_predictive_analysis_master")
print("✅ Table fully compiled!")
